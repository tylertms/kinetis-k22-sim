  function(kinetis_k22_add_test test_name test_source test_labels)
    set(test_target "kinetis_k22_${test_name}_test")
    add_executable(${test_target} ${test_source} ${ARGN})
    target_compile_features(${test_target} PRIVATE c_std_11)
    target_include_directories(${test_target} PRIVATE tests/support tests src)
    target_link_libraries(${test_target} PRIVATE kinetis_k22_simulator)
    kinetis_k22_enable_strict_warnings(${test_target})
    add_test(NAME ${test_target} COMMAND ${test_target})
    set_tests_properties(${test_target} PROPERTIES LABELS "${test_labels}")
  endfunction()

  function(kinetis_k22_enable_allocation_failure test_target)
    if(NOT MSVC)
      target_sources(${test_target} PRIVATE tests/support/allocation_failure.c)
      target_compile_definitions(${test_target}
                                 PRIVATE K22_TEST_ALLOCATION_FAILURE=1)
      target_link_options(${test_target} PRIVATE -Wl,--wrap=malloc
                          -Wl,--wrap=calloc -Wl,--wrap=realloc)
    endif()
  endfunction()
